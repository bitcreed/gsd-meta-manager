3. Create `src/config.rs` with these exact types and functions (per D-01, D-02, D-04):
```rust
use serde::{Serialize, Deserialize};
use std::collections::HashMap;
use std::path::{Path, PathBuf};

#[derive(Debug, Serialize, Deserialize)]
pub struct Config {
    pub version: u32,
    pub projects: HashMap<String, RegisteredProject>,
    #[serde(default)]
    pub preferences: Preferences,
}

#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct RegisteredProject {
    pub path: PathBuf,
    pub added: String,
}

#[derive(Debug, Serialize, Deserialize, Default)]
pub struct Preferences {}

impl Config {
    pub fn default_path() -> PathBuf {
        dirs::config_dir()
            .unwrap_or_else(|| PathBuf::from(".config"))
            .join("gsd-manager")
            .join("config.json")
    }
    pub fn new() -> Self {
        Config { version: 1, projects: HashMap::new(), preferences: Preferences::default() }
    }
}

pub fn load_config(path: &Path) -> anyhow::Result<Config> // returns new Config if file missing
pub fn save_config(config: &Config, path: &Path) -> anyhow::Result<()> // atomic write via tempfile+rename
```
- `load_config`: If file doesn't exist, return `Config::new()`. If file exists, deserialize JSON. Use `anyhow::Context` for error messages.
- `save_config`: Create parent dir with `create_dir_all`, write to `NamedTempFile::new_in(parent_dir)`, call `.persist(path)` for atomic rename.
